# Szablony skryptów dla różnych formatów obsługiwanych przez ScriptDB

# ---------------------------------------------------
# PYTHON (.py)
# Używane do skryptów ogólnego przeznaczenia i logiki biznesowej.
# ---------------------------------------------------
print("Rozpoczynanie skryptu Python...")
try:
    # Tutaj umieść swoją logikę.
    data = "Przykład danych"
    print(f"Przetwarzanie danych: {data}")
except Exception as e:
    print(f"Wystąpił błąd: {e}")

# ---------------------------------------------------
# SQL (.sql)
# Używane do zapytań do baz danych.
# ---------------------------------------------------
-- Wybierz wszystkie rekordy z tabeli użytkowników
SELECT * FROM users WHERE status = 'active';

-- Dodaj nowy rekord do tabeli produktów
INSERT INTO products (name, price) VALUES ('Laptop', 1200.00);

# ---------------------------------------------------
# RUBY (.rb)
# Używane do skryptów zorientowanych na przetwarzanie danych w Ruby.
# ---------------------------------------------------
puts "Rozpoczynanie skryptu Ruby..."

# Definiowanie i użycie funkcji
def greet(name)
  puts "Witaj, #{name}!"
end

greet("Świat")

# ---------------------------------------------------
# BASH (.sh / .bash)
# Używane do automatyzacji zadań systemowych w środowisku Unix/Linux.
# ---------------------------------------------------
#!/bin/bash
echo "Rozpoczynanie skryptu Bash..."

# Sprawdzanie statusu systemu
df -h

# Przykład pętli
for i in {1..3}; do
  echo "Iteracja numer $i"
done

# ---------------------------------------------------
# DATA (.dat)
# Używane dla formatów danych niestandardowych lub prostych list.
# ---------------------------------------------------
# Data format: Key|Value
user_id|1001
username|testuser
email|test@example.com

# ---------------------------------------------------
# VBSCRIPT (.vbs)
# Używane do skryptów automatyzacji w środowisku Windows.
# ---------------------------------------------------
Dim WshShell
Set WshShell = CreateObject("WScript.Shell")
WScript.Echo "Rozpoczynanie skryptu VBScript..."

' Wykonanie komendy
WshShell.Run "cmd /c echo Halo Świat", 1, True

' ---------------------------------------------------
# BATCH (.bat)
# Używane do prostych skryptów uruchamiania w środowisku Windows.
# ---------------------------------------------------
@echo off
echo Rozpoczynam skrypt BATCH...
echo Dane przetworzone pomyślnie.
pause

# ---------------------------------------------------
# POWERSHELL (.ps1)
# Używane do zaawansowanej automatyzacji w środowisku Windows.
# ---------------------------------------------------
Write-Host "Rozpoczynanie skryptu PowerShell..."

# Uzyskanie listy procesów
Get-Process | Select-Object Name, ID | Out-GridView

# Uruchomienie polecenia systemu
Start-Sleep -Seconds 2
Write-Host "Skrypt PowerShell zakończony."
